async function doSearch() { const raw = document.getElementById('inp').value.trim(); if (!raw) return; const btn = document.getElementById('btn'); const div = document.getElementById('result'); btn.disabled = true; div.innerHTML = '
🔍 조회 중...
'; try { let parts = raw.split(/\s+/); let sido = ''; for (const [k, v] of Object.entries(SIDO)) { if (parts[0] === k || parts[0].startsWith(k)) { sido = v; parts = parts.slice(1); break; } } const name = parts.join(' '); // 1단계: 병원 검색 const r1 = await fetch(`${WORKER}?q=${encodeURIComponent(name)}&sido=${encodeURIComponent(sido)}`); if (!r1.ok) { div.innerHTML = '
공공데이터 서버 응답 지연
잠시 후 다시 시도해 주세요.
'; return; } const j1 = await r1.json(); const items = j1?.response?.body?.items?.item || []; const list = Array.isArray(items) ? items : (items ? [items] : []); if (!list.length) { div.innerHTML = '
검색 결과 없음

'; return; } // 2단계: ykiho 디코딩 (워커를 통해 심평원 Ajax 우회 크롤링) let html = ''; for (const h of list.slice(0, 5)) { // 주소 창에서 깔끔하게 매칭하기 위해 암호화 ykiho 그대로 전달 const r2 = await fetch(`${WORKER}?ykiho=${encodeURIComponent(h.ykiho)}`); let ykiho = '-'; if (r2.ok) { const j2 = await r2.json(); ykiho = j2.ykiho || '-'; } const addr = h.addr1 || ''; html += `

`; } div.innerHTML = html; } catch(e) { div.innerHTML = `

`; } finally { btn.disabled = false; } }
