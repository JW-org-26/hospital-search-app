export default { async fetch(req) { const cors = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS' }; if (req.method === 'OPTIONS') return new Response(null, { headers: cors }); const url = new URL(req.url); const name = url.searchParams.get('q') || ''; const commonHeaders = { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36', 'Accept': 'application/json, text/javascript, */*; q=0.01', 'Accept-Language': 'ko-KR,ko;q=0.9,en-US;q=0.8,en;q=0.7', 'X-Requested-With': 'XMLHttpRequest', 'Referer': 'https://www.hira.or.kr/ra/hosp/getHealthMap.do' }; try { // 💡 심평원 서버가 요구하는 필수 파라미터 규격을 완벽히 일치시킵니다. const formData = new URLSearchParams(); formData.append('searchKeyword', name); formData.append('searchFlag', 'A'); // 심평원 엔진에서 전체 조회를 활성화하는 마스터 플래그 const r = await fetch('https://www.hira.or.kr/ra/hosp/getHealthMapList.do', { method: 'POST', headers: { ...commonHeaders, 'Content-Type': 'application/x-www-form-urlencoded; charset=UTF-8' }, body: formData.toString() }); const resData = await r.json(); const items = []; // 💡 심평원 실제 API의 응답 키는 대문자 'List'로 들어옵니다. (resData.List) const list = resData.List || resData.list || resData.resList || []; for (let i = 0; i < list.length; i++) { if (items.length >= 5) break; // 최대 5개 노출 const h = list[i]; // 💡 심평원 실시간 JSON 규격에 박혀있는 필드명을 명확하게 맵핑합니다. // yadmNm: 병원명, ykiho: 8자리 진짜 요양기관기호, addr: 실제 주소 let hospName = h.yadmNm || h.YADMNM || ''; let ykihoNum = h.ykiho || h.YKIHO || ''; let address = h.addr || h.ADDR || '주소 정보 없음'; // HTML 태그 잔여물 제거 if (hospName) hospName = hospName.replace(/<\/?[^>]+(>|$)/g, "").trim(); if (address) address = address.replace(/<\/?[^>]+(>|$)/g, "").trim(); if (hospName && ykihoNum) { items.push({ yadmNm: hospName, ykiho: ykihoNum, // 이제 복잡한 2차 우회 없이 8자리 기호가 다이렉트로 꽂힙니다. addr1: address }); } } return new Response(JSON.stringify({ items }), { headers: { ...cors, 'Content-Type': 'application/json' } }); } catch(e) { return new Response(JSON.stringify({ error: e.message, items: [] }), { headers: { ...cors, 'Content-Type': 'application/json' } }); } } }
